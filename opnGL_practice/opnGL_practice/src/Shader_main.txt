#shader vertex
#version 430 core

layout(location = 0) in vec3 positions;
layout(location = 1) in vec3 aNormal;
layout(location = 2) in vec2 aTextCoords;

uniform mat4 model;
uniform mat4 view;
uniform mat4 projection;

out vec3 FragPos;
out vec3 Normal;
out vec2 TextCoords;

void main()
{
	FragPos = vec3(model * vec4(positions, 1.0));
	//Normal = aNormal;
	Normal = mat3(transpose(inverse(model))) * aNormal;
	TextCoords = aTextCoords;
	
	gl_Position = projection * view  * vec4(FragPos, 1.0);

};



#shader fragment
#version 430 core

struct Material
{
	sampler2D diffuse;
	sampler2D specular;
	sampler2D emission;
	float shininess;
};

struct Light
{
	vec3 ambient;
	vec3 diffusion;
	vec3 specular;
	vec3 position;
};

out vec4 color;
uniform vec3 objectColor;
uniform vec3 lightColor;
uniform vec3 lightPos;
uniform vec3 viewPos;
uniform Material material;
uniform Light light;
uniform float time;

in vec3 Normal;
in vec3 FragPos;
in vec2 TextCoords;

void main()
{
	float ambientStrength = 0.1;
	float specularStrength = 0.5;

	//----------1. Ambient ----------
	//vec3 ambient = ambientStrength * lightColor;
	//vec3 ambient = light.ambient * material.ambient;
	vec3 ambient = light.ambient * vec3(texture(material.diffuse, TextCoords));

	//----------2. Diffusion ----------
	//normal dir vector
	vec3 norm = normalize(Normal);

	//light direction vector
	vec3 lightDir = normalize(lightPos - FragPos); 

	//diffusion
	float diff = max(dot(norm, lightDir), 0.0);
	//vec3 diffuse = diff * lightColor;
	//vec3 diffuse = (diff * material.diffusion) * light.diffusion;
	vec3 diffuse = light.diffusion * diff * vec3(texture(material.diffuse, TextCoords));


	//-----------3. Specular ---------------
	//camera view direction
	vec3 viewDir = normalize(viewPos - FragPos);

	//reflect dir vector
	vec3 reflectDir = reflect(-lightDir, norm);


	//specular
	//float spec = pow(max(dot(viewDir, reflectDir), 0.0), 32);
	float spec = pow(max(dot(viewDir, reflectDir), 0.0), material.shininess);

	//vec3 specular = specularStrength * spec * lightColor;
	vec3 specular = light.specular * spec * vec3(texture(material.specular, TextCoords)); //steel boundary specular map
	//vec3 specular = light.specular * spec * (vec3(1.0f) - vec3(texture(material.specular, TextCoords))); // inverting the color values of the specular map
	//vec3 specular = light.specular * spec * vec3(texture(material.specular, TextCoords)) * vec3(1.0, 0.0, 0.0); //red boundary

	//-----------4. Emission Map ---------------
	vec3 emission = vec3(0.0);
	if(texture(material.specular, TextCoords).r == 0.0) /*rough check for blackbox inside spec texture */
	{	
		//emission = vec3(texture(material.emission, TextCoords)); //still
	
		/*some extra fun stuff with "time uniform" */
        emission = vec3(texture(material.emission, TextCoords + vec2(0.0,time)));   /*moving */
        //emission = emission * (sin(time) * 0.5 + 0.5) * 2.0;  //fading
	}

	//result
	//vec3 result = (ambient + diffuse + specular) * objectColor;
	//vec3 result = (ambient + diffuse + specular);
	vec3 result = (ambient + diffuse + specular + emission);
	color = vec4(result, 1.0);
	
	//color = vec4(lightColor * objectColor, 1.0f);
};
